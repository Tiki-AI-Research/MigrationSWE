type f2pCreateCall struct {
	tag    string
	draft  *bool
	body   *string
	commit *string
}

type f2pUpdateCall struct {
	id     int64
	draft  *bool
	body   *string
	commit *string
}

type f2pReleases struct {
	getByTagResult *ReleaseData
	getByTagErr    error
	createCalls    []f2pCreateCall
	updateCalls    []f2pUpdateCall
	listAssets     []ReleaseAsset
	deleted        []int64
	uploadErr      error
	uploadCalls    int
}

func (f *f2pReleases) Create(tag string, body *string, commitHash *string, discussionCategory *string, draft *bool, makeLatest *string, name *string, prerelease *bool) (ReleaseData, error) {
	f.createCalls = append(f.createCalls, f2pCreateCall{tag, draft, body, commitHash})
	return ReleaseData{ID: 100, UploadURL: "https://uploads.example.com", TagName: tag, HTMLURL: "https://example.com/100"}, nil
}

func (f *f2pReleases) Update(id int64, tag string, body *string, commitHash *string, discussionCategory *string, draft *bool, makeLatest *string, name *string, prerelease *bool) (ReleaseData, error) {
	f.updateCalls = append(f.updateCalls, f2pUpdateCall{id, draft, body, commitHash})
	return ReleaseData{ID: id, UploadURL: "https://uploads.example.com", TagName: tag, HTMLURL: "https://example.com/100"}, nil
}

func (f *f2pReleases) GetByTag(tag string) (*ReleaseData, error) { return f.getByTagResult, f.getByTagErr }

func (f *f2pReleases) GenerateReleaseNotes(tag string, previousTag *string, targetCommitish *string) (GenerateReleaseNotesResponse, error) {
	return GenerateReleaseNotesResponse{Body: "generated"}, nil
}

func (f *f2pReleases) ListReleases() (*ListReleasesResponse, error) {
	return &ListReleasesResponse{}, nil
}

func (f *f2pReleases) ListArtifactsForRelease(releaseId int64) ([]ReleaseAsset, error) {
	return f.listAssets, nil
}

func (f *f2pReleases) DeleteArtifact(assetId int64) error {
	f.deleted = append(f.deleted, assetId)
	return nil
}

func (f *f2pReleases) UploadArtifact(assetUrl string, contentLength int64, contentType string, file io.Reader, name string, releaseId int64) (UploadArtifactResponse, error) {
	f.uploadCalls++
	if f.uploadErr != nil {
		return UploadArtifactResponse{}, f.uploadErr
	}
	return UploadArtifactResponse{BrowserDownloadURL: "https://downloads.example.com/" + name}, nil
}

type f2pOutputs struct {
	release *ReleaseData
	urls    map[string]string
}

func (o *f2pOutputs) ApplyReleaseData(data ReleaseData) {
	copied := data
	o.release = &copied
}

func (o *f2pOutputs) ApplyAssetUrls(urls map[string]string) { o.urls = urls }

type f2pUploader struct {
	err   error
	calls int
}

func (u *f2pUploader) UploadArtifacts(artifacts []Artifact, releaseId int64, uploadUrl string) (map[string]string, error) {
	u.calls++
	if u.err != nil {
		return nil, u.err
	}
	urls := map[string]string{}
	for _, artifact := range artifacts {
		urls[artifact.Name] = "https://downloads.example.com/" + artifact.Name
	}
	return urls, nil
}

type f2pDestroyer struct{ calls int }

func (d *f2pDestroyer) DestroyArtifacts(releaseId int64) error { d.calls++; return nil }

type f2pSkipper struct {
	skip bool
	err  error
}

func (s *f2pSkipper) ShouldSkip() (bool, error) { return s.skip, s.err }

type f2pInputGetter struct {
	values  map[string]string
	bools   map[string]bool
	boolErr map[string]error
}

func (g *f2pInputGetter) GetInput(name string) string {
	if v, ok := g.values[name]; ok {
		return v
	}
	return ""
}

func (g *f2pInputGetter) GetBooleanInput(name string) (bool, error) {
	if err, ok := g.boolErr[name]; ok {
		return false, err
	}
	return g.bools[name], nil
}

type f2pStubGlobber struct {
	result   []string
	err      error
	patterns []string
}

func (g *f2pStubGlobber) Glob(pattern string) ([]string, error) {
	g.patterns = append(g.patterns, pattern)
	return g.result, g.err
}

type f2pRecordGlobber struct {
	artifact    string
	contentType string
	result      []Artifact
}

func (g *f2pRecordGlobber) GlobArtifactString(artifact string, contentType string, errorsFailBuild bool) ([]Artifact, error) {
	g.artifact = artifact
	g.contentType = contentType
	return g.result, nil
}

func f2pLoad(t *testing.T, values map[string]string, bools map[string]bool, ctx ActionContext, globber ArtifactGlobber) *Inputs {
	t.Helper()
	if globber == nil {
		globber = &f2pRecordGlobber{result: []Artifact{}}
	}
	merged := map[string]string{"token": "token"}
	for key, value := range values {
		merged[key] = value
	}
	inputs, err := LoadInputs(&f2pInputGetter{values: merged, bools: bools}, globber, ctx)
	if err != nil {
		t.Fatalf("LoadInputs returned error: %v", err)
	}
	return inputs
}

func f2pContext() ActionContext {
	return ActionContext{Ref: "refs/tags/context-tag", Owner: "context-owner", Repo: "context-repo"}
}

func f2pRead(t *testing.T, path string) string {
	t.Helper()
	data, err := os.ReadFile(path)
	if err != nil {
		t.Fatalf("read %s: %v", path, err)
	}
	return string(data)
}

func TestF2P025(t *testing.T) {

		t.Fatalf("action.yml runs.image is not Dockerfile")

		t.Fatalf("Dockerfile does not build the Go binary")
	}
	if !strings.Contains(docker, "ENTRYPOINT") {
		t.Fatalf("Dockerfile does not set an ENTRYPOINT")
	}
}

// ---------------------------------------------------------------------------
// F2P-031 (R030) compiled Go binary does not delegate to Node.
// ---------------------------------------------------------------------------

func TestF2P031(t *testing.T) {
	mod := f2pRead(t, "go.mod")
	if strings.Contains(mod, "github.com/actions/") || strings.Contains(mod, "@actions/") {
		t.Fatalf("go.mod depends on Node action modules")
	}
	if !strings.Contains(mod, "github.com/bmatcuk/doublestar/v4") {
		t.Fatalf("go.mod missing the expected doublestar dependency")
	}

	entries, err := filepath.Glob("*.go")
	if err != nil {
		t.Fatal(err)
	}
	if len(entries) == 0 {
		t.Fatal("no Go production sources in module root")
	}
	for _, name := range entries {
		if strings.HasSuffix(name, "_test.go") {
			continue
		}
		src := f2pRead(t, name)
		if strings.Contains(src, "\"os/exec\"") || strings.Contains(src, "exec.Command") {
			t.Fatalf("%s spawns subprocesses instead of handling the behavior in-process", name)
		}
		if strings.Contains(src, "dist/index.js") || strings.Contains(src, "node_modules") {
			t.Fatalf("%s references the Node bundle", name)
		}
	}
}

// ---------------------------------------------------------------------------
// F2P-043 (R047) immutableCreate request sequence.
// ---------------------------------------------------------------------------

func TestF2P043(t *testing.T) {
	releases := &f2pReleases{}
	outputs := &f2pOutputs{}
	inputs := &Inputs{Tag: "v1", ImmutableCreate: true, CreatedDraft: false}
	action := NewAction(inputs, outputs, releases, &f2pUploader{}, &f2pDestroyer{}, &f2pSkipper{})
	if err := action.Perform(); err != nil {
		t.Fatal(err)
	}

	if len(releases.createCalls) != 1 || releases.createCalls[0].draft == nil || !*releases.createCalls[0].draft {
		t.Fatalf("expected a draft create first, got %+v", releases.createCalls)
	}
	if len(releases.updateCalls) != 1 {
		t.Fatalf("expected one publish update, got %+v", releases.updateCalls)
	}
	update := releases.updateCalls[0]
	if update.body != nil || update.commit != nil || update.draft == nil || *update.draft {
		t.Fatalf("publish update should omit body/commit and set draft=false: %+v", update)
	}
	if outputs.release == nil || outputs.release.ID != update.id {
		t.Fatalf("outputs were not taken from the publish response: %+v", outputs.release)
	}
}
